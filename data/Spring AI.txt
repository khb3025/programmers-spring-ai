# 컨테이너 내부에서 모델 다운로드
# 컴퓨터 자원에 맞는 모델 선택:
# - llama3 (8B): 약 4.7GB, 8GB+ RAM 권장
# - gemma:2b: 약 1.7GB, 4GB+ RAM 권장 (가벼운 모델)
# - phi3: 약 2.2GB, 4GB+ RAM 권장 (가벼운 모델)
docker compose exec ollama ollama pull llama3
docker compose exec ollama ollama pull gemma:2b
docker compose exec ollama ollama pull phi3

# - qwen2.5:0.5b (0.5B): 약 398MB, 2GB+ RAM 권장 (초경량, 기본 연동 테스트 추천)
# - tinyllama (1.1B): 약 637MB, 2GB+ RAM 권장 (초경량 채팅 모델)
# - gemma3:1b (1B): 약 815MB, 4GB+ RAM 권장 (경량 모델)
# - qwen2.5:1.5b (1.5B): 약 986MB, 4GB+ RAM 권장 (한국어 채팅 추천)
